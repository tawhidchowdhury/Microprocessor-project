
.MODEL SMALL

.STACK 100H

.DATA
;Login Data for senders and receivers
sender_ids      DB "s1","s2","s3"
sender_pwds     DB "ab1","cd2","ef3"
receiver_ids    DB "r1","r2","r3"
receiver_pwds   DB "rc1","rc2","rc3"

;message buffers
message_buffer  DB 100 DUP('$')
reply_buffer    DB 100 DUP('$')
secret_key_buffer DB 100 DUP('$')    ; Buffer to store the secret key
current_sender DB '$'    ; Store the current sender's ID
current_receiver DB '$'  ; Store the current receiver's ID

;Temporary 
username_input  DB 3 DUP('$')
password_input  DB 3 DUP('$')
receiver_choice DB 3 DUP('$')
 
; Add encryption method prompt
msg_encryption_prompt DB 13, 10, "Choose encryption method (1=Caesar, 2=Atbash, 3=ROT13): $"
encryption_method DB 0 ; Store the selected encryption method
msg_encrypted_reply DB 13, 10, "Encrypted Reply Message: $"
msg_encrypted_message DB 13, 10, "Encrypted Message: $"

;UI prompts
msg_login_prompt    DB "Login as Sender (s1/s2/s3): $"
msg_pwd_prompt      DB 13, 10, "Password: $"
msg_wrong_cred      DB 13, 10, "Invalid credentials. Terminating.$"
msg_receiver_prompt DB 13, 10, "Send message to (r1/r2/r3): $"
msg_input_prompt    DB 13, 10, "Enter your message: $"
msg_reply_prompt    DB 13, 10, "Do you want to reply? (y/n): $"
msg_reply_input     DB 13, 10, "Enter reply message: $"
msg_show_reply      DB 13, 10, "Sender replied: $"
msg_received        DB 13, 10, "Message received: $"   ; Add this new message indicator
receiver_login_msg  DB 13, 10, "Login as Receiver (r1/r2/r3): $"
msg_login_error     DB 13, 10, "Invalid login. Please try again.$"
view_reply_prompt   DB 13, 10, "Login as Sender to view reply: $"
msg_reply_received  DB 13, 10, "Reply received: $"   ; Add this new message indicator
press_key_msg       DB 13, 10, "Press 'q' to quit or any other key to continue...$" ; Add this new message
msg_secret_key_prompt DB 13, 10, "Enter secret key for this message: $"
msg_enter_key_prompt DB 13, 10, "Enter secret key to view message: $"
msg_wrong_key DB 13, 10, "Wrong key! Message is protected.$"
hidden_msg DB "xxxxx$"

;receiver flags
receiver1_flag DB 0    ; 0 = False, 1 = True
receiver2_flag DB 0
receiver3_flag DB 0
msg_unauthorized DB 13, 10, "Unauthorized! This message is not for you.$"

; Add sender flags
sender1_flag DB 0    ; 0 = False, 1 = True
sender2_flag DB 0
sender3_flag DB 0
msg_unauthorized_reply DB 13, 10, "Unauthorized! This reply is not for you.$"

original_sender DB 'x$'    ; keeping the number of original sender (1,2,3)
can_reply DB 0    ; flag to track if receiver can reply (0=no, 1=yes)

.CODE
MAIN PROC
; Initialize DS
Mov AX, @DATA
Mov DS, AX

START:
;clearing screen first 

CALL CLEAR_SCREEN

; first sender login 

Lea DX, msg_login_prompt
Mov AH, 9
INT 21H

; get sender login and send message 

CALL GET_SENDER_LOGIN
CALL SEND_MESSAGE

; clearing screen for receiver

CALL CLEAR_SCREEN

; receiver login

Lea DX, receiver_login_msg
Mov AH, 9
INT 21H

; get receiver login

CALL GET_RECEIVER_LOGIN

; display message to receiver

CALL DISPLAY_MESSAGE

;Handle reply

CALL HANDLE_REPLY
CMP AL, 'y'            ; checking if receiver chose to reply
JE CONTINUE_PROGRAM    ; If y continue
JMP EXIT_PROGRAM    

CONTINUE_PROGRAM:
; Clear screen for final sender login
CALL CLEAR_SCREEN

; Show prompt for sender to view reply
Lea DX, view_reply_prompt
Mov AH, 9
INT 21H

; final sender login to view reply
CALL GET_SENDER_LOGIN

; show the reply
CALL DISPLAY_REPLY

EXIT_PROGRAM:
; Exit
Mov AX, 4C00H
INT 21H

MAIN ENDP

; New procedure to clear screen
CLEAR_SCREEN PROC
    Mov AX, 0600H    ; Scroll window up
    Mov BH, 07H      ; Normal attributes
    Mov CX, 0000H    ; Upper left corner (0,0)
    Mov DX, 184FH    ; Lower right corner (24,79)
    INT 10H
    
    ; Reset cursor position
    Mov AH, 02H
    Mov BH, 00H
    Mov DX, 0000H
    INT 10H
    RET
CLEAR_SCREEN ENDP

; new procedure to handle receiver login

GET_RECEIVER_LOGIN PROC
    PUSH AX
    PUSH DX
    
GET_RECEIVER_INPUT:
    Lea DX, username_input
    CALL GET_STRING

    Lea DX, msg_pwd_prompt
    Mov AH, 9
    INT 21H

    Lea DX, password_input
    CALL GET_STRING

    CALL VALIDATE_RECEIVER
    CMP AL, 0
    JE INVALID_RECEIVER_LOGIN

    POP DX
    POP AX
    RET

INVALID_RECEIVER_LOGIN:
    Lea DX, msg_wrong_cred
    Mov AH, 9
    INT 21H
    JMP GET_RECEIVER_INPUT
GET_RECEIVER_LOGIN ENDP


; new procedure to display message

DISPLAY_MESSAGE PROC
    PUSH AX
    PUSH DX

   
;clear screen first
    CALL CLEAR_SCREEN

;reset reply permission
    Mov can_reply, 0

;check if this receiver is authorized
    CALL CHECK_RECEIVER_AUTH
    CMP AL, 0        ; AL will be 0 if unauthorized
    JE UNAUTHORIZED_ACCESS

;If authorized, continue with key check
    Lea DX, msg_enter_key_prompt
    Mov AH, 9
    INT 21H

;Get key from receiver
    Lea SI, password_input    ; Use password_input as temporary buffer
    Mov CX, 0
GET_KEY:
    Mov AH, 1
    INT 21H
    CMP AL, 13               ; Check for Enter key
    JE CHECK_KEY
    Mov [SI], AL
    INC SI
    INC CX
    JMP GET_KEY

CHECK_KEY:
    Mov byte ptr [SI], '$'   ; Terminate input string

; Compare entered key with stored key
    Lea SI, password_input
    Lea DI, secret_key_buffer
    CALL COMPARE_KEYS
    CMP AL, 1                ; AL=1 if keys match
    JNE WRONG_KEY

; Set reply permission since key matched
    Mov can_reply, 1

; Decrypt the message
    Lea SI, message_buffer
    Lea DI, message_buffer
    CALL DECRYPT_MESSAGE

; Display message received header
    Lea DX, msg_received
    Mov AH, 9
    INT 21H
; Display the actual message
    Lea DX, message_buffer
    Mov AH, 9
    INT 21H
    JMP DISPLAY_DONE

UNAUTHORIZED_ACCESS:
    Lea DX, msg_unauthorized
    Mov AH, 9
    INT 21H
    JMP DISPLAY_DONE

WRONG_KEY:
; Show error message
    Lea DX, msg_wrong_key
    Mov AH, 9
    INT 21H

; Display xxxxx instead of message
    Lea DX, hidden_msg
    Mov AH, 9
    INT 21H

DISPLAY_DONE:
; Add newline
    Mov AH, 2
    Mov DL, 13
    INT 21H
    Mov DL, 10
    INT 21H

    POP DX
    POP AX
    RET
DISPLAY_MESSAGE ENDP

; New procedure to handle reply
HANDLE_REPLY PROC
    PUSH DX
    
;Check if receiver is allowed to reply
    CMP can_reply, 0
    JE NO_REPLY_PERMISSION
    
    Lea DX, msg_reply_prompt
    Mov AH, 9
    INT 21H
    
    Mov AH, 1
    INT 21H
    PUSH AX      ; Save the user's choice (y/n)
    
    CMP AL, 'y'
    JNE NO_REPLY_CHOSEN
    
;Get reply message
    Lea DX, msg_reply_input
    Mov AH, 9
    INT 21H
    
    Lea DX, reply_buffer
    CALL GET_STRING

;Encrypt the reply message
    Lea SI, reply_buffer
    Lea DI, reply_buffer
    CALL ENCRYPT_MESSAGE

;Display the encrypted reply message to the receiver
    Lea DX, msg_encrypted_reply
    Mov AH, 9
    INT 21H
    Lea DX, reply_buffer
    Mov AH, 9
    INT 21H

;Clear screen for sender login
    CALL CLEAR_SCREEN
    
;Call LOGIN_SENDER procedure to handle the complete login process
    CALL LOGIN_SENDER
    
NO_REPLY_PERMISSION:
    ; Skip reply prompt if not authorized
    Mov AL, 'n'   ; Set response as if user chose not to reply
    PUSH AX
    JMP NO_REPLY_CHOSEN
    
NO_REPLY_CHOSEN:
    POP AX       ; Restore the user's choice to AL
    
    POP DX
    RET         ; AL contains 'y' or 'n'
HANDLE_REPLY ENDP

LOGIN_SENDER PROC
    Lea DX, msg_login_prompt
    Mov AH, 9
    INT 21H

    Lea DX, username_input
    CALL GET_STRING

    Lea DX, msg_pwd_prompt
    Mov AH, 9
    INT 21H

    Lea DX, password_input
    CALL GET_STRING

;Validate sender credentials
    CALL VALIDATE_SENDER
    CMP AL, 0
    JE INVALID_CREDS

;Check if this sender is authorized
    CALL CHECK_SENDER_AUTH
    CMP AL, 0
    JE UNAUTHORIZED_SENDER
    
;Clear screen and show reply
    CALL CLEAR_SCREEN
    
;Display reply header
    Lea DX, msg_reply_received
    Mov AH, 9
    INT 21H

;Decrypt the reply message
    Lea SI, reply_buffer
    Lea DI, message_buffer
    CALL DECRYPT_MESSAGE

;Display the decrypted reply
    Lea DX, message_buffer
    Mov AH, 9
    INT 21H
    
;Add newline
    Mov AH, 2
    Mov DL, 13
    INT 21H
    Mov DL, 10
    INT 21H

;Show quit prompt
    Lea DX, press_key_msg
    Mov AH, 9
    INT 21H
    
;Get user input
    Mov AH, 1
    INT 21H
    
;Check if 'q' was pressed
    CMP AL, 'q'
    JE DO_EXIT
    
    RET

INVALID_CREDS:
    Lea DX, msg_wrong_cred
    Mov AH, 9
    INT 21H
;Return with error status
    Mov AL, 0
    RET

UNAUTHORIZED_SENDER:
    Lea DX, msg_unauthorized_reply
    Mov AH, 9
    INT 21H
    RET

DO_EXIT:
    Mov AX, 4C00H
    INT 21H    ; Exit to DOS
LOGIN_SENDER ENDP

LOGIN_RECEIVER PROC
    Lea DX, msg_login_prompt
    Mov AH, 9
    INT 21H

    Lea DX, username_input
    CALL GET_STRING

    Lea DX, msg_pwd_prompt
    Mov AH, 9
    INT 21H

    Lea DX, password_input
    CALL GET_STRING

; Validate receiver credentials
    CALL VALIDATE_RECEIVER
    CMP AL, 0
    JE INVALID_CREDENTIALS
    
; Add a newline before showing message
    Mov AH, 2
    Mov DL, 13
    INT 21H
    Mov DL, 10
    INT 21H
    
; Display received message indicator
    Lea DX, msg_received
    Mov AH, 9
    INT 21H
    
    RET
LOGIN_RECEIVER ENDP

SEND_MESSAGE PROC
; Prompt for receiver selection
    Lea DX, msg_receiver_prompt
    Mov AH, 9
    INT 21H

; Get receiver choice
    Lea DX, receiver_choice
    CALL GET_STRING

; Store current sender and receiver IDs
    Lea SI, username_input
    Mov AL, [SI+1]     ; Get the number after 's'
    Mov current_sender, AL  ; Store current sender

    Lea SI, receiver_choice
    Mov AL, [SI+1]     ; Get the number after 'r'
    Mov current_receiver, AL  ; Store current receiver
;Store the original sender number
    Lea SI, username_input
    Mov AL, [SI+1]     ; Get the number after 's'
    Mov original_sender, AL  ; Store it

; Reset all receiver flags
    Mov receiver1_flag, 0
    Mov receiver2_flag, 0
    Mov receiver3_flag, 0

;Set the appropriate receiver flag
    Lea SI, receiver_choice
    Mov AL, [SI]       ; Get 'r'
    CMP AL, 'r'
    JNE GET_MESSAGE    ; If not 'r', skipping setting flags

    Mov AL, [SI+1]     ; Get the number after 'r'
    CMP AL, '1'
    JE SET_R1
    CMP AL, '2'
    JE SET_R2
    CMP AL, '3'
    JE SET_R3
    JMP GET_MESSAGE

SET_R1:
    Mov receiver1_flag, 1
    JMP GET_MESSAGE
SET_R2:
    Mov receiver2_flag, 1
    JMP GET_MESSAGE
SET_R3:
    Mov receiver3_flag, 1

GET_MESSAGE:
;Prompt for encryption method
    Lea DX, msg_encryption_prompt
    Mov AH, 9
    INT 21H

; Get encryption method
    Mov AH, 1
    INT 21H
    SUB AL, '0'        ; Convert ASCII to number
    Mov encryption_method, AL

;Prompt for message input
    Lea DX, msg_input_prompt
    Mov AH, 9
    INT 21H

;Get the message
    Lea DX, message_buffer
    CALL GET_STRING

;Encrypt the message
    Lea SI, message_buffer
    Lea DI, message_buffer
    CALL ENCRYPT_MESSAGE

; Display the encrypted message to the sender
    Lea DX, msg_encrypted_message
    Mov AH, 9
    INT 21H
    Lea DX, message_buffer
    Mov AH, 9
    INT 21H

;Prompt for secret key
    Lea DX, msg_secret_key_prompt
    Mov AH, 9
    INT 21H

;Get secret key
    Lea DX, secret_key_buffer
    CALL GET_STRING

; Reset all sender flags
    Mov sender1_flag, 0
    Mov sender2_flag, 0
    Mov sender3_flag, 0

;Set the appropriate sender flag
    Lea SI, username_input
    Mov AL, [SI+1]     ; Get the number after 's'
    CMP AL, '1'
    JE SET_S1
    CMP AL, '2'
    JE SET_S2
    CMP AL, '3'
    JE SET_S3
    JMP CONTINUE_SEND

SET_S1:
    Mov sender1_flag, 1
    JMP CONTINUE_SEND
SET_S2:
    Mov sender2_flag, 1
    JMP CONTINUE_SEND
SET_S3:
    Mov sender3_flag, 1

CONTINUE_SEND:
    RET
SEND_MESSAGE ENDP

; Ensure the receiver sees the message after logging in
VIEW_MESSAGE PROC
    ; Clear screen first for better visibility
    Mov AH, 0   ; Set video mode
    Mov AL, 3   ; Text mode 80x25
    INT 10H
; Display the message
    Lea DX, message_buffer
    Mov AH, 9
    INT 21H
    
;newlines for better formatting
    Mov AH, 2       ; Function to display character
    Mov DL, 13      ; Carriage return
    INT 21H
    Mov DL, 10      ; Line feed
    INT 21H
    Mov DL, 10      ; Another line feed for extra space
    INT 21H
    RET
VIEW_MESSAGE ENDP

REPLY_MESSAGE PROC
    Lea DX, msg_reply_prompt
    Mov AH, 9
    INT 21H

    Mov AH, 1
    INT 21H
    CMP AL, 'y'
    JNE NO_REPLY

; Get reply message
    Lea DX, msg_reply_input
    Mov AH, 9
    INT 21H

    Lea DX, reply_buffer
    CALL GET_STRING

NO_REPLY:
    RET
REPLY_MESSAGE ENDP

VIEW_REPLY PROC
    Lea DX, reply_buffer
    Mov AH, 9
    INT 21H
    RET
VIEW_REPLY ENDP

GET_STRING PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH SI

    Mov CX, 0
    Mov SI, DX    ; SI points to buffer

READ_CHAR:
    Mov AH, 1   ; Read character
    INT 21H
    CMP AL, 13     ; Enter key?
    JE END_INPUT
    Mov [SI], AL
    INC SI
    INC CX
    CMP CX, 100  ; Limit input to 100 characters
    JGE END_INPUT
    JMP READ_CHAR

END_INPUT:
    Mov [SI], '$'; Terminate with $
    POP SI
    POP CX
    POP BX
    POP AX
    RET
GET_STRING ENDP

; Fixed VALIDATE_SENDER procedure to handle all sender IDs correctly
VALIDATE_SENDER PROC
    PUSH BX
    PUSH CX
    PUSH DX
    PUSH SI
    PUSH DI

; Get the sender number from input (after 's')
    Lea SI, username_input
    Mov AL, [SI+1]  ; Get the number after 's'
    SUB AL, '1'  ; Convert ASCII number to 0-based index
    
    ; Calculate offset into sender_pwds multiply by 3 since each pwd is 3 chars
    Mov BL, 3
    MUL BL        ; AX = AL * BL
    Mov BX, AX    ; Save offset in BX
    
    ; Compare password
    Lea SI, password_input
    Lea DI, sender_pwds
    ADD DI, BX         ; Point to correct password
    
    Mov CX, 3         ; Compare 3 characters
CHECK_PWD:
    Mov AL, [SI]
    CMP AL, [DI]
    JNE INVALID_PWD
    INC SI
    INC DI
    LOOP CHECK_PWD
    
    Mov AL, 1         ; Password matches
    JMP EXIT_VS
    
INVALID_PWD:
    Mov AL, 0         ; Password doesn't match
    
EXIT_VS:
    POP DI
    POP SI
    POP DX
    POP CX
    POP BX
    RET
VALIDATE_SENDER ENDP

VALIDATE_RECEIVER PROC
    PUSH SI
    PUSH DI
    PUSH BX
    PUSH CX

; Get input ID (first character)
    Lea SI, username_input
    Mov AL, [SI+1]     ; Get the number after 'r'
    SUB AL, '1'        ; Convert ASCII number to 0-based index
    
; Calculate offset into receiver_pwds multiply by 3 since each pwd is 3 chars
    Mov BL, 3
    MUL BL            ; AX = AL * BL
    Mov BX, AX        ; Save offset in BX
    
    ; Compare password
    Lea SI, password_input
    Lea DI, receiver_pwds
    ADD DI, BX         ; Point to correct password
    
    Mov CX, 3         ; Compare 3 characters
CHECK_PWD_RECEIVER:
    Mov AL, [SI]
    CMP AL, [DI]
    JNE INVALID_RECEIVER
    INC SI
    INC DI
    LOOP CHECK_PWD_RECEIVER
    
    Mov AL, 1         ; Password matches
    JMP EXIT_VALIDATE_RECEIVER
    
INVALID_RECEIVER:
    Mov AL, 0         ; Password doesn't match
    
EXIT_VALIDATE_RECEIVER:
    POP CX
    POP BX
    POP DI
    POP SI
    RET
VALIDATE_RECEIVER ENDP

STRCMP PROC
    PUSH AX
    PUSH SI
    PUSH DI

    Mov AL, 1          ; Assume match

COMPARE_LOOP:
    Mov BL, [SI]       ; Load byte from source
    CMP BL, '$'        ; Check for string terminator
    JE END_STR_CMP
    CMP BL, [DI]       ; Compare with destination
    JNE NOT_EQUAL
    INC SI             ; Move to next byte in source
    INC DI             ; Move to next byte in destination
    JMP COMPARE_LOOP

END_STR_CMP:
    CMP [DI], '$'      ; Ensure both strings end at the same point
    JNE NOT_EQUAL
    JMP DONE_CMP

NOT_EQUAL:
    Mov AL, 0          ; Set result to 0 (not equal)

DONE_CMP:
    POP DI
    POP SI
    POP AX
    RET
STRCMP ENDP

    ; Check if this sender is authorized to view reply
    CALL CHECK_SENDER_AUTH
    CMP AL, 0
    JE UNAUTH_REPLY

    ; Clear screen first
    CALL CLEAR_SCREEN

    ; Display reply header
    Lea DX, msg_reply_received
    Mov AH, 9
    INT 21H

    ; Decrypt the reply message
    Lea SI, reply_buffer
    Lea DI, message_buffer

    JMP REPLY_DONE

UNAUTH_REPLY:
; Display unauthorized access message
    Lea DX, msg_unauthorized_reply
    Mov AH, 9
    INT 21H

REPLY_DONE:
; Add newline for formatting
    Mov AH, 2

    Mov DL, 10
    INT 21H

    Lea DX, press_key_msg
    Mov AH, 9
    INT 21H

    ; Wait for any key
    Mov AH, 1

    RET
DISPLAY_REPLY ENDP

COMPARE_KEYS PROC
    PUSH BX
    PUSH CX
    PUSH DX
    
    Mov AL, 1      ; Assume match initially
    
COMPARE_LOOP_KEY:
    Mov BL, [SI]
    Mov BH, [DI]
    
    CMP BL, '$'  ; Check if we reached end of first string
    JE CHECK_END_KEY
    
    CMP BL, BH  ; Compare characters
    JNE NO_MATCH_KEY
    
    INC SI
    INC DI
    JMP COMPARE_LOOP_KEY
    
CHECK_END_KEY:
    CMP BH, '$' ; Make sure second string also ended
    JNE NO_MATCH_KEY
    JMP MATCH_KEY
    
NO_MATCH_KEY:
    Mov AL, 0   ; Set no match
    
MATCH_KEY:
    POP DX
    POP CX
    POP BX
    RET           ; AL contains 1 if match, 0 if no match
COMPARE_KEYS ENDP

; New procedure to check if the current receiver is authorized
CHECK_RECEIVER_AUTH PROC
    PUSH BX
    PUSH DX
    
    ; Get receiver ID from username_input
    Lea SI, username_input
    Mov AL, [SI]       ; Get first char ('r')
    CMP AL, 'r'
    JNE NOT_AUTHORIZED
    
    Mov AL, [SI+1]     ; Get the number after 'r'
    CMP AL, '1'
    JE CHECK_R1
    CMP AL, '2'
    JE CHECK_R2
    CMP AL, '3'
    JE CHECK_R3
    JMP NOT_AUTHORIZED
    
CHECK_R1:
    Mov AL, receiver1_flag
    JMP CHECK_AUTH_DONE
CHECK_R2:
    Mov AL, receiver2_flag
    JMP CHECK_AUTH_DONE
CHECK_R3:
    Mov AL, receiver3_flag
    JMP CHECK_AUTH_DONE
    
NOT_AUTHORIZED:
    Mov AL, 0
    
CHECK_AUTH_DONE:
    POP DX
    POP BX
    RET    ; AL contains 1 if authorized, 0 if not
CHECK_RECEIVER_AUTH ENDP

; New procedure to check sender authorization
CHECK_SENDER_AUTH PROC
    PUSH BX
    PUSH DX
    
; Get sender ID from username_input
    Lea SI, username_input
    Mov AL, [SI]       ; Get first char ('s')
    CMP AL, 's'
    JNE NOT_AUTH_SENDER
    
    Mov AL, [SI+1]     ; Get the number after 's'
    
;Compare with stored sender
    CMP AL, current_sender
    JNE NOT_AUTH_SENDER
    
; If we get here, this is the authorized sender
    Mov AL, 1
    JMP CHECK_SENDER_DONE
    
NOT_AUTH_SENDER:
    Mov AL, 0
    
CHECK_SENDER_DONE:
    POP DX
    POP BX
    RET    ; AL contains 1 if authorized, 0 if not
CHECK_SENDER_AUTH ENDP

; CHANGE
; encrypt message procedure
ENCRYPT_MESSAGE PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX

    Mov AL, encryption_method
    CMP AL, 1
    JE ENCRYPT_CAESAR
    CMP AL, 2
    JE ENCRYPT_ATBASH
    CMP AL, 3
    JE ENCRYPT_ROT13
    JMP ENCRYPT_DONE

ENCRYPT_CAESAR:
    ; Caesar cipher (shift by 3)
    Mov CX, 0
CAESAR_LOOP:
    Mov AL, [SI]
    CMP AL, '$'
    JE ENCRYPT_DONE

    ; Check if character is uppercase
    CMP AL, 'A'
    JB SKIP_CAESAR      ; Skip if less than 'A'
    CMP AL, 'Z'
    JBE CAESAR_UPPERCASE ; If between 'A' and 'Z', process as uppercase

    ; Check if character is lowercase
    CMP AL, 'a'
    JB SKIP_CAESAR      ; Skip if less than 'a'
    CMP AL, 'z'
    JBE CAESAR_LOWERCASE ; If between 'a' and 'z', process as lowercase

SKIP_CAESAR:
    ; Non-alphabetic character, copy as is
    Mov [DI], AL
    INC SI
    INC DI
    JMP CAESAR_LOOP

CAESAR_UPPERCASE:
    ; Apply Caesar cipher for uppercase letters
    ADD AL, 3
    CMP AL, 'Z'
    JLE CONTINUE_CAESAR
    SUB AL, 26          ; Wrap around if past 'Z'
CONTINUE_CAESAR:
    Mov [DI], AL
    INC SI
    INC DI
    JMP CAESAR_LOOP

CAESAR_LOWERCASE:
    ; Apply Caesar cipher for lowercase letters
    ADD AL, 3
    CMP AL, 'z'
    JLE CONTINUE_CAESAR2
    SUB AL, 26          ; Wrap around if past 'z'
CONTINUE_CAESAR2:
    Mov [DI], AL
    INC SI
    INC DI
    JMP CAESAR_LOOP

ENCRYPT_ATBASH:
    Mov CX, 0
ATBASH_LOOP:
    Mov AL, [SI]        ; Load the current character
    CMP AL, '$'         ; Check for end of string
    JE ENCRYPT_DONE

    ; Check if character is uppercase
    CMP AL, 'A'
    JB SKIP_ATBASH      ; Skip if less than 'A'
    CMP AL, 'Z'
    JBE ATBASH_UPPERCASE ; If between 'A' and 'Z', process as uppercase

    ; Check if character is lowercase
    CMP AL, 'a'
    JB SKIP_ATBASH      ; Skip if less than 'a'
    CMP AL, 'z'
    JBE ATBASH_LOWERCASE ; If between 'a' and 'z', process as lowercase

SKIP_ATBASH:
    ; Non-alphabetic character, copy as is
    Mov [DI], AL
    INC SI
    INC DI
    JMP ATBASH_LOOP

ATBASH_UPPERCASE:
    ; Reverse uppercase letters
    Mov BL, 'Z'
    SUB AL, 'A'         ; Normalize to 0-based index
    Mov CL, AL          ; Save the normalized value
    Mov AL, BL
    SUB AL, CL          ; Reverse the position
    Mov [DI], AL        ; Store the result
    INC SI
    INC DI
    JMP ATBASH_LOOP

ATBASH_LOWERCASE:
    ; Reverse lowercase letters
    Mov BL, 'z'
    SUB AL, 'a'         ; Normalize to 0-based index
    Mov CL, AL          ; Save the normalized value
    Mov AL, BL
    SUB AL, CL          ; Reverse the position
    Mov [DI], AL        ; Store the result
    INC SI
    INC DI
    JMP ATBASH_LOOP

ENCRYPT_ROT13:
    ; ROT13 cipher
    Mov CX, 0
ROT13_LOOP:
    Mov AL, [SI]
    CMP AL, '$'
    JE ENCRYPT_DONE

    ; Check if character is uppercase
    CMP AL, 'A'
    JB ROT13_SKIP
    CMP AL, 'Z'
    JBE ROT13_UPPERCASE

    ; Check if character is lowercase
    CMP AL, 'a'
    JB ROT13_SKIP
    CMP AL, 'z'
    JBE ROT13_LOWERCASE

ROT13_SKIP:
    ; Non-alphabetic character, copy as is
    Mov [DI], AL
    INC SI
    INC DI
    JMP ROT13_LOOP

ROT13_UPPERCASE:
    ; Apply ROT13 for uppercase letters
    ADD AL, 13
    CMP AL, 'Z'
    JLE ROT13_CONTINUE
    SUB AL, 26
ROT13_CONTINUE:
    Mov [DI], AL
    INC SI
    INC DI
    JMP ROT13_LOOP

ROT13_LOWERCASE:
    ; Apply ROT13 for lowercase letters
    ADD AL, 13
    CMP AL, 'z'
    JLE ROT13_CONTINUE2
    SUB AL, 26
ROT13_CONTINUE2:
    Mov [DI], AL
    INC SI
    INC DI
    JMP ROT13_LOOP

ENCRYPT_DONE:
    ; End of encryption
    POP DX
    POP CX
    POP BX
    POP AX
    RET
ENCRYPT_MESSAGE ENDP

; decrypt message procedure

DECRYPT_MESSAGE PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX

    Mov AL, encryption_method
    CMP AL, 1
    JE DECRYPT_CAESAR
    CMP AL, 2
    JE DECRYPT_ATBASH
    CMP AL, 3
    JE DECRYPT_ROT13
    JMP DECRYPT_DONE

DECRYPT_CAESAR:
    ; Caesar cipher (shift by -3)
    Mov CX, 0
CAESAR_LOOP_DEC:
    Mov AL, [SI]
    CMP AL, '$'
    JE DECRYPT_DONE

    ; Check if character is uppercase
    CMP AL, 'A'
    JB SKIP_CAESAR_DEC      ; Skip if less than 'A'
    CMP AL, 'Z'
    JBE CAESAR_UPPERCASE_DEC ; If between 'A' and 'Z', process as uppercase

    JBE CAESAR_LOWERCASE_DEC ; If between 'a' and 'z', process as lowercase

SKIP_CAESAR_DEC:
    ; Non-alphabetic character, copy as is
    Mov [DI], AL
    INC SI
    INC DI
    JMP CAESAR_LOOP_DEC

CAESAR_UPPERCASE_DEC:
    ; Apply Caesar cipher for uppercase letters
    SUB AL, 3
    CMP AL, 'A'
    JGE CONTINUE_CAESAR_DEC
    ADD AL, 26          ; Wrap around if before 'A'
CONTINUE_CAESAR_DEC:
    Mov [DI], AL
    INC SI
    INC DI
    JMP CAESAR_LOOP_DEC

CAESAR_LOWERCASE_DEC:
    ; Apply Caesar cipher for lowercase letters
    SUB AL, 3
    CMP AL, 'a'
    JGE CONTINUE_CAESAR2_DEC
    ADD AL, 26          ; Wrap around if before 'a'
CONTINUE_CAESAR2_DEC:
    Mov [DI], AL
    INC SI
    INC DI
    JMP CAESAR_LOOP_DEC

DECRYPT_ATBASH:
    Mov CX, 0
ATBASH_LOOP_DEC:
    Mov AL, [SI]    ; Load the current character
    CMP AL, '$'         ; Check for end of string
    JE DECRYPT_DONE

    ; Check if character is uppercase
    CMP AL, 'A'
    JB SKIP_ATBASH_DEC  ; Skip if less than 'A'
    CMP AL, 'Z'
    JBE ATBASH_UPPERCASE_DEC ; If between 'A' and 'Z', process as uppercase

    ; Check if character is lowercase
    CMP AL, 'a'
    JB SKIP_ATBASH_DEC  ; Skip if less than 'a'
    CMP AL, 'z'
    JBE ATBASH_LOWERCASE_DEC ; If between 'a' and 'z', process as lowercase

SKIP_ATBASH_DEC:
    ; Non-alphabetic character, copy as is
    Mov [DI], AL
    INC SI
    INC DI
    JMP ATBASH_LOOP_DEC

ATBASH_UPPERCASE_DEC:
    ; Reverse uppercase letters
    Mov BL, 'Z'
    SUB AL, 'A'    ; Normalize to 0-based index
    Mov CL, AL     ; Save the normalized value
    Mov AL, BL

    Mov [DI], AL     ; Store the result
    INC SI
    INC DI
    JMP ATBASH_LOOP_DEC

ATBASH_LOWERCASE_DEC:
    ; Reverse lowercase letters
    Mov BL, 'z'
    SUB AL, 'a'    ; Normalize to 0-based index
    Mov CL, AL    ; Save the normalized value
    Mov AL, BL
    SUB AL, CL    ; Reverse the position
    Mov [DI], AL    ; Store the result
    INC SI
    INC DI
    JMP ATBASH_LOOP_DEC

DECRYPT_ROT13:
    ; ROT13 cipher (same as encryption)
    Mov CX, 0
ROT13_LOOP_DEC:
    Mov AL, [SI]     ; Load the current character
    CMP AL, '$'         ; Check for end of string
    JE DECRYPT_DONE

    ; Check if character is uppercase
    CMP AL, 'A'
    JB ROT13_SKIP_DEC   ; Skip if less than 'A'
    CMP AL, 'Z'
    JBE ROT13_UPPERCASE_DEC ; If between 'A' and 'Z', process as uppercase

    ; Check if character is lowercase
    CMP AL, 'a'
    JB ROT13_SKIP_DEC   ; Skip if less than 'a'
    CMP AL, 'z'
    JBE ROT13_LOWERCASE_DEC ; If between 'a' and 'z', process as lowercase

ROT13_SKIP_DEC:
    ; Non-alphabetic character, copy as is
    Mov [DI], AL
    INC SI
    INC DI
    JMP ROT13_LOOP_DEC

ROT13_UPPERCASE_DEC:
    ; Apply ROT13 for uppercase letters
    ADD AL, 13
    CMP AL, 'Z'
    JLE ROT13_CONTINUE_DEC
    SUB AL, 26          ; Wrap around if past 'Z'
ROT13_CONTINUE_DEC:
